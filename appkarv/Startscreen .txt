import React, { useState, useEffect } from 'react';

export default function App() {
  const [isLoaded, setIsLoaded] = useState(false);
  const [stars, setStars] = useState([]);
  const [floatingIcons, setFloatingIcons] = useState([]);

  useEffect(() => {
    // تفعيل تأثير الظهور السلس بعد تحميل المكون
    setTimeout(() => setIsLoaded(true), 100);

    // توليد النجوم العشوائية
    const generatedStars = Array.from({ length: 150 }).map((_, i) => ({
      id: i,
      left: `${Math.random() * 100}%`,
      top: `${Math.random() * 100}%`,
      size: Math.random() * 3 + 1,
      animationDuration: `${Math.random() * 3 + 2}s`,
      animationDelay: `${Math.random() * 2}s`,
    }));
    setStars(generatedStars);

    // توليد الأيقونات الإسلامية العشوائية (أهلة ونجوم ثمانية وفوانيس)
    const icons = ['crescent', 'star', 'lantern'];
    const generatedIcons = Array.from({ length: 15 }).map((_, i) => ({
      id: i,
      type: icons[Math.floor(Math.random() * icons.length)],
      left: `${Math.random() * 90 + 5}%`,
      top: `${Math.random() * 90 + 5}%`,
      size: Math.random() * 20 + 15,
      animationDuration: `${Math.random() * 10 + 10}s`,
      animationDelay: `${Math.random() * 5}s`,
      opacity: Math.random() * 0.3 + 0.1,
    }));
    setFloatingIcons(generatedIcons);
  }, []);

  // دوال لرسم الأيقونات الإسلامية كـ SVG
  const renderIcon = (type, size) => {
    switch (type) {
      case 'crescent':
        return (
          <svg width={size} height={size} viewBox="0 0 24 24" fill="currentColor" xmlns="http://www.w3.org/2000/svg">
            <path d="M21 12.79A9 9 0 1 1 11.21 3 7 7 0 0 0 21 12.79z" />
          </svg>
        );
      case 'star': // نجمة إسلامية (تقريبية)
        return (
          <svg width={size} height={size} viewBox="0 0 24 24" fill="currentColor" xmlns="http://www.w3.org/2000/svg">
            <path d="M12 2L14.8 9.2H22L16.2 13.8L18.4 21L12 16.6L5.6 21L7.8 13.8L2 9.2H9.2L12 2Z" />
          </svg>
        );
      case 'lantern':
        return (
          <svg width={size} height={size} viewBox="0 0 24 24" fill="currentColor" xmlns="http://www.w3.org/2000/svg">
            <path d="M10 2L14 2L15 6L9 6L10 2Z" />
            <path d="M8 7L16 7L18 17L6 17L8 7Z" />
            <path d="M9 18L15 18L14 22L10 22L9 18Z" />
          </svg>
        );
      default:
        return null;
    }
  };

  return (
    <div className="relative min-h-screen w-full bg-[#080514] overflow-hidden flex flex-col items-center justify-center font-reem">
      
      {/* تضمين خط كوفي ريم من جوجل */}
      <style>
        {`
          @import url('https://fonts.googleapis.com/css2?family=Reem+Kufi:wght@400;500;600;700&display=swap');
          
          .font-reem {
            font-family: 'Reem Kufi', sans-serif;
          }

          .animate-fade-in-up {
            animation: fadeInUp 1.5s cubic-bezier(0.2, 0.8, 0.2, 1) forwards;
          }

          @keyframes fadeInUp {
            0% { opacity: 0; transform: translateY(40px); }
            100% { opacity: 1; transform: translateY(0); }
          }

          .twinkle {
            animation: twinkle var(--duration) ease-in-out infinite alternate;
            animation-delay: var(--delay);
          }

          @keyframes twinkle {
            0% { opacity: 0.2; transform: scale(0.8); }
            100% { opacity: 1; transform: scale(1.2); box-shadow: 0 0 10px rgba(255, 255, 255, 0.8); }
          }

          .float {
            animation: float var(--duration) ease-in-out infinite alternate;
            animation-delay: var(--delay);
          }

          @keyframes float {
            0% { transform: translateY(0) rotate(0deg); }
            100% { transform: translateY(-30px) rotate(15deg); }
          }
          
          .ring-glow {
            box-shadow: 0 0 20px currentColor, inset 0 0 20px currentColor;
          }
        `}
      </style>

      {/* خلفية النجوم */}
      {stars.map((star) => (
        <div
          key={star.id}
          className="absolute rounded-full bg-white twinkle"
          style={{
            left: star.left,
            top: star.top,
            width: `${star.size}px`,
            height: `${star.size}px`,
            '--duration': star.animationDuration,
            '--delay': star.animationDelay,
          }}
        />
      ))}

      {/* الأيقونات الإسلامية العائمة في الخلفية */}
      {floatingIcons.map((icon) => (
        <div
          key={icon.id}
          className="absolute text-[#8b5cf6] float"
          style={{
            left: icon.left,
            top: icon.top,
            opacity: icon.opacity,
            '--duration': icon.animationDuration,
            '--delay': icon.animationDelay,
          }}
        >
          {renderIcon(icon.type, icon.size)}
        </div>
      ))}

      {/* المحتوى الرئيسي */}
      <div 
        className={`z-10 flex flex-col items-center transition-all duration-1000 ${
          isLoaded ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-10'
        }`}
      >
        {/* الحاوية الخاصة بالأيقونة وحلقات زحل */}
        <div className="relative flex items-center justify-center w-72 h-72 mb-8">
          
          {/* حلقات زحل (Saturn Rings) */}
          <div 
            className="absolute inset-0 flex items-center justify-center"
            style={{ 
              transform: 'rotateX(75deg) rotateY(-15deg)', 
              transformStyle: 'preserve-3d' 
            }}
          >
            {/* الحلقة الأولى (خارجية سريعة) */}
            <div className="absolute w-[360px] h-[360px] rounded-full border-[3px] border-[#00f2fe] border-t-transparent animate-[spin_4s_linear_infinite] ring-glow text-[#00f2fe]/40 opacity-70"></div>
            
            {/* الحلقة الثانية (وسطى معاكسة) */}
            <div className="absolute w-[310px] h-[310px] rounded-full border-[2px] border-[#a18cd1] border-b-transparent animate-[spin_6s_linear_infinite_reverse] ring-glow text-[#a18cd1]/50 opacity-80"></div>
            
            {/* الحلقة الثالثة (داخلية بطيئة) */}
            <div className="absolute w-[270px] h-[270px] rounded-full border-[1px] border-[#fbc2eb] animate-[spin_10s_linear_infinite] opacity-40"></div>
          </div>

          {/* أيقونة التطبيق */}
          <img 
            src="1000021748.png" 
            alt="أيقونة إمساكية كرار" 
            className="w-48 h-48 rounded-full z-10 shadow-[0_0_50px_rgba(139,92,246,0.6)] object-cover relative"
          />
        </div>

        {/* النصوص */}
        <div className="text-center animate-fade-in-up" style={{ animationDelay: '0.3s', opacity: 0 }}>
          <h1 className="text-5xl md:text-6xl font-bold text-transparent bg-clip-text bg-gradient-to-r from-[#00f2fe] via-[#fbc2eb] to-[#a18cd1] drop-shadow-[0_0_10px_rgba(255,255,255,0.3)] mb-4">
            إمساكية كرار
          </h1>
          <h2 className="text-2xl md:text-3xl text-gray-300/90 font-medium tracking-wide drop-shadow-md">
            علي المرتضى والمهدي
          </h2>
        </div>
      </div>

      {/* التوقيع السفلي */}
      <div 
        className={`absolute bottom-8 text-gray-500/40 text-sm md:text-base font-medium tracking-widest transition-opacity duration-1000 delay-1000 ${
          isLoaded ? 'opacity-100' : 'opacity-0'
        }`}
      >
        عبودي
      </div>

    </div>
  );
}
