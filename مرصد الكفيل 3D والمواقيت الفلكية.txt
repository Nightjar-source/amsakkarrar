// دالة تقسيم الشهر للكشيدة المتصلة الفاخرة
const getKashidaMonthParts = (monthName) => {
  const clean = (monthName || "رمضان").replace(/[\u064B-\u065F\u0651\u0654]/g, '');
  if (clean.includes("محرم")) return { pre: "محـ", post: "ـرم" };
  if (clean.includes("صفر")) return { pre: "صـ", post: "ـفر" };
  if (clean.includes("ربيع الأول")) return { pre: "ربيـ", post: "ـع الأول" };
  if (clean.includes("ربيع الآخر") || clean.includes("ربيع الثاني")) return { pre: "ربيـ", post: "ـع الآخر" };
  if (clean.includes("جمادى الأولى")) return { pre: "جمـ", post: "ـادى الأولى" };
  if (clean.includes("جمادى الآخرة") || clean.includes("جمادى الثانية")) return { pre: "جمـ", post: "ـادى الآخرة" };
  if (clean.includes("رجب")) return { pre: "رجـ", post: "ـب" };
  if (clean.includes("شعبان")) return { pre: "شعبـ", post: "ـان" };
  if (clean.includes("رمضان")) return { pre: "رمضـ", post: "ـان" };
  if (clean.includes("شوال")) return { pre: "شـ", post: "ـوال" };
  if (clean.includes("القعدة")) return { pre: clean.replace("القعدة", "القعـ"), post: "ـدة" };
  if (clean.includes("الحجة")) return { pre: clean.replace("الحجة", "الحجـ"), post: "ـة" };
  return { pre: clean.slice(0, -1) + "ـ", post: "ـ" + clean.slice(-1) };
};

const formatArcHour = (rawHourFloat) => {
  if (rawHourFloat === undefined || isNaN(rawHourFloat)) return { t: '00:00', ap: 'صباحاً', hhmm24: '00:00' };
  let local = ((rawHourFloat % 24) + 24) % 24;
  let h = Math.floor(local);
  let m = Math.floor((local - h) * 60 + 0.5);
  if (m >= 60) { h = (h + 1) % 24; m = 0; }
  const ap = h >= 12 ? 'مساءً' : 'صباحاً';
  const h12 = h % 12 || 12;
  return {
    t: `${h12}:${String(m).padStart(2, '0')}`,
    ap,
    hhmm24: `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}`
  };
};

const base64ToArrayBuffer = (base64) => {
  const binaryString = window.atob(base64);
  const len = binaryString.length;
  const bytes = new Uint8Array(len);
  for (let i = 0; i < len; i++) {
    bytes[i] = binaryString.charCodeAt(i);
  }
  return bytes.buffer;
};

const pcmToWav = (pcm16, sampleRate = 24000) => {
  const buffer = new ArrayBuffer(44 + pcm16.length * 2);
  const view = new DataView(buffer);
  const writeString = (offset, string) => {
    for (let i = 0; i < string.length; i++) {
      view.setUint8(offset + i, string.charCodeAt(i));
    }
  };
  writeString(0, 'RIFF');
  view.setUint32(4, 36 + pcm16.length * 2, true);
  writeString(8, 'WAVE');
  writeString(12, 'fmt ');
  view.setUint32(16, 16, true);
  view.setUint16(20, 1, true);
  view.setUint16(22, 1, true);
  view.setUint32(24, sampleRate, true);
  view.setUint32(28, sampleRate * 2, true);
  view.setUint16(32, 2, true);
  view.setUint16(34, 16, true);
  writeString(36, 'data');
  view.setUint32(40, pcm16.length * 2, true);
  for (let i = 0; i < pcm16.length; i++) {
    view.setInt16(44 + i * 2, pcm16[i], true);
  }
  return new Blob([buffer], { type: 'audio/wav' });
};

/* ========================================================================
   📐 المحرك الفلكي الرياضي المستقل (Offline Astronomical Math Engine)
   ======================================================================== */
const DTR = Math.PI / 180.0; 
const RTD = 180.0 / Math.PI;

const calculateDistance = (lat1, lon1, lat2, lon2) => {
  const R = 6371; 
  const dLat = (lat2 - lat1) * DTR; 
  const dLon = (lon2 - lon1) * DTR;
  const a = Math.sin(dLat / 2) * Math.sin(dLat / 2) + 
            Math.cos(lat1 * DTR) * Math.cos(lat2 * DTR) * Math.sin(dLon / 2) * Math.sin(dLon / 2);
  return R * (2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a)));
};

// حساب اتجاه القبلة والمسافة إلى الكعبة المشرفة أوفلاين
const calculateQiblaAndMoon = (lat, lon, jd) => {
  const kaabaLat = 21.4225;
  const kaabaLon = 39.8262;
  const dLon = (kaabaLon - lon) * DTR;
  const latRad = lat * DTR;
  const kaabaLatRad = kaabaLat * DTR;
  
  const y = Math.sin(dLon);
  const x = Math.cos(latRad) * Math.tan(kaabaLatRad) - Math.sin(latRad) * Math.cos(dLon);
  let qibla = Math.atan2(y, x) * RTD;
  qibla = (qibla + 360) % 360;
  
  const distToKaaba = Math.round(calculateDistance(lat, lon, kaabaLat, kaabaLon));

  // حساب طور القمر الفلكي الدقيق (Synodic Month = 29.530588853 days)
  const moonCycle = ((jd - 2451550.1) / 29.530588853) % 1.0;
  const normalizedPhase = moonCycle < 0 ? moonCycle + 1.0 : moonCycle;
  const illumination = Math.round((1 - Math.cos(normalizedPhase * 2 * Math.PI)) / 2 * 100);
  const moonAgeDays = (normalizedPhase * 29.530588853).toFixed(1);
  
  let phaseName = "محاق";
  let phaseSymbol = "🌑";
  if (normalizedPhase > 0.02 && normalizedPhase < 0.23) { phaseName = "هلال أول الشهر (متزايد)"; phaseSymbol = "🌒"; }
  else if (normalizedPhase >= 0.23 && normalizedPhase <= 0.27) { phaseName = "تربيع أول"; phaseSymbol = "🌓"; }
  else if (normalizedPhase > 0.27 && normalizedPhase < 0.48) { phaseName = "أحدب متزايد"; phaseSymbol = "🌔"; }
  else if (normalizedPhase >= 0.48 && normalizedPhase <= 0.52) { phaseName = "بدر مكتمل"; phaseSymbol = "🌕"; }
  else if (normalizedPhase > 0.52 && normalizedPhase < 0.73) { phaseName = "أحدب متناقص"; phaseSymbol = "🌖"; }
  else if (normalizedPhase >= 0.73 && normalizedPhase <= 0.77) { phaseName = "تربيع أخير"; phaseSymbol = "🌗"; }
  else if (normalizedPhase > 0.77 && normalizedPhase < 0.98) { phaseName = "هلال آخر الشهر (متناقص)"; phaseSymbol = "🌘"; }

  return { qibla: qibla.toFixed(1), distToKaaba, phaseName, phaseSymbol, illumination, normalizedPhase, moonAgeDays };
};

// دالة هندسية فلكية متقدمة لرسم مسار الجزء المضاء من القمر والهلال بدقة بصرية عالية (Spherical Terminator Projection)
const buildAccurateMoonLitPath = (phase, r = 18) => {
  const p = ((phase % 1) + 1) % 1;
  if (p < 0.012 || p > 0.988) return ""; // محاق
  if (p > 0.488 && p < 0.512) {
    // بدر مكتمل دائري
    return `M 0 ${-r} A ${r} ${r} 0 1 1 0 ${r} A ${r} ${r} 0 1 1 0 ${-r} Z`;
  }
  const isWaxing = p < 0.5;
  const angle = p * 2 * Math.PI;
  const cosTerm = Math.cos(angle);
  const rx = Math.max(0.15, Math.abs(cosTerm) * r);

  // في النصف الأول (هلال/تربيع/أحدب متزايد) الإضاءة من اليمين، وفي النصف الثاني (متناقص) من اليسار
  const outerSweep = isWaxing ? 1 : 0;
  let innerSweep;
  if (isWaxing) {
    innerSweep = p < 0.25 ? 0 : 1; // هلال مقعر عند p < 0.25 وأحدب محدب عند p > 0.25
  } else {
    innerSweep = p < 0.75 ? 0 : 1; // أحدب متناقص ثم هلال آخر الشهر
  }

  return `M 0 ${-r} A ${r} ${r} 0 0 ${outerSweep} 0 ${r} A ${rx.toFixed(3)} ${r} 0 0 ${innerSweep} 0 ${-r} Z`;
};

// تحويل التاريخ الهجري (سنة، شهر، يوم) إلى كائن تاريخ ميلادي لاستعراض أي شهر أو سنة هجرية سابقة أو قادمة
const hijriToGregorianDate = (hYear, hMonth, hDay = 1, isShiaMode = true, adjustmentDays = 0) => {
  const leapYears = [2, 5, 7, 10, 13, 16, 18, 21, 24, 26, 29];
  const fullCycles = Math.floor((hYear - 1) / 30);
  const yearInCycle = ((hYear - 1) % 30) + 1;
  let daysAccum = fullCycles * 10631;
  for (let yr = 1; yr < yearInCycle; yr++) {
    daysAccum += leapYears.includes(yr) ? 355 : 354;
  }
  const monthLengths = [30, 29, 30, 29, 30, 29, 30, 29, 30, 29, 30, 30];
  for (let mn = 1; mn < hMonth; mn++) {
    daysAccum += monthLengths[mn - 1];
  }
  daysAccum += (hDay - 1);
  const shiaOffset = isShiaMode ? -1 : 0;
  const jd = daysAccum + 1948439.5 - shiaOffset - adjustmentDays;
  const unixMs = (jd - 2440587.5) * 86400000;
  const approxDate = new Date(unixMs);
  approxDate.setHours(12, 0, 0, 0);

  // ضبط دقيق لليوم الأول من الشهر الهجري المطلوب
  for (let offset = -4; offset <= 4; offset++) {
    const testDate = new Date(approxDate);
    testDate.setDate(testDate.getDate() + offset);
    const checkH = getHijriDateSafe(testDate, isShiaMode, adjustmentDays);
    if (checkH.monthNum === hMonth && checkH.day === 1) {
      return testDate;
    }
  }
  return approxDate;
};

const fixAngle = (a) => { let b = a - 360.0 * Math.floor(a / 360.0); return b < 0 ? b + 360.0 : b; };
const fixHour = (a) => { let b = a - 24.0 * Math.floor(a / 24.0); return b < 0 ? b + 24.0 : b; };
const dsin = (d) => Math.sin(d * DTR); 
const dcos = (d) => Math.cos(d * DTR); 
const darcsin = (x) => Math.asin(x) * RTD; 
const darccos = (x) => Math.acos(x) * RTD; 
const darctan2 = (y, x) => Math.atan2(y, x) * RTD;

const getJulianDay = (year, month, day) => {
  if (month <= 2) { year -= 1; month += 12; }
  const A = Math.floor(year / 100); 
  const B = 2 - A + Math.floor(A / 4);
  return Math.floor(365.25 * (year + 4716)) + Math.floor(30.6001 * (month + 1)) + day + B - 1524.5;
};

const getSolarPosition = (jd) => {
  const D = jd - 2451545.0;
  const g = fixAngle(357.529 + 0.98560028 * D);
  const q = fixAngle(280.459 + 0.98564736 * D);
  const L = fixAngle(q + 1.915 * dsin(g) + 0.02 * dsin(2 * g));
  const e = 23.439 - 0.00000036 * D;
  const RA = darctan2(dcos(e) * dsin(L), dcos(L)) / 15.0;
  const declination = darcsin(dsin(e) * dsin(L));
  const eqt = q / 15.0 - fixHour(RA);
  return { declination, eqt };
};

const getHourAngle = (angle, declination, latitude) => {
  const cosHA = (-dsin(angle) - dsin(declination) * dsin(latitude)) / (dcos(declination) * dcos(latitude));
  if (cosHA > 1 || cosHA < -1) return NaN; 
  return darccos(cosHA) / 15.0;
};

const formatTimeOutput = (time, tzOffset, lon, manualOffsetMinutes = 0, use24Hour = false) => {
  if (isNaN(time) || time === null || isNaN(tzOffset)) return "--:--";
  let localTime = fixHour(time + parseFloat(tzOffset) - (parseFloat(lon) / 15.0) + (manualOffsetMinutes / 60.0));
  let h = Math.floor(localTime);
  let m = Math.floor((localTime - h) * 60 + 0.5);
  if (m >= 60) { h = (h + 1) % 24; m = 0; }
  
  if (use24Hour) {
    return `${h.toString().padStart(2, '0')}:${m.toString().padStart(2, '0')}`;
  }
  let suffix = h >= 12 ? 'م' : 'ص';
  let h12 = h % 12 || 12;
  return `${h12.toString().padStart(2, '0')}:${m.toString().padStart(2, '0')} ${suffix}`;
};

/* ========================================================================
   🌙 خوارزمية التقويم الهجري الاصطلاحي (Tabular Hijri Algorithm)
   ======================================================================== */
const getHijriDateSafe = (dateObj, isShiaMode, adjustmentDays = 0) => {
  const year = dateObj.getFullYear();
  const month = dateObj.getMonth() + 1;
  const day = dateObj.getDate();
  
  let y = year;
  let m = month;
  if (m <= 2) { y -= 1; m += 12; }
  const A = Math.floor(y / 100);
  const B = 2 - A + Math.floor(A / 4);
  const jd = Math.floor(365.25 * (y + 4716)) + Math.floor(30.6001 * (m + 1)) + day + B - 1524.5;
  
  const shiaOffset = isShiaMode ? -1 : 0;
  const jdHijri = jd - 1948439.5 + 0.5 + shiaOffset + adjustmentDays;
  
  const cycles = Math.floor(jdHijri / 10631);
  const Rem = Math.floor(jdHijri % 10631);
  const leapYears = [2, 5, 7, 10, 13, 16, 18, 21, 24, 26, 29];
  
  let yearInCycle = 1;
  let daysAccumulated = 0;
  for (let yr = 1; yr <= 30; yr++) {
    const yearLen = leapYears.includes(yr) ? 355 : 354;
    if (daysAccumulated + yearLen > Rem) {
      yearInCycle = yr;
      break;
    }
    daysAccumulated += yearLen;
  }
  
  const daysThisYear = Rem - daysAccumulated + 1;
  const hYear = cycles * 30 + yearInCycle;
  
  let hMonth = 1;
  let hDay = daysThisYear;
  const monthLengths = [30, 29, 30, 29, 30, 29, 30, 29, 30, 29, 30, 30];
  const isLeapYear = leapYears.includes(yearInCycle);
  
  for (let mn = 1; mn <= 12; mn++) {
    const mLen = (mn === 12) ? (isLeapYear ? 30 : 29) : monthLengths[mn - 1];
    if (hDay <= mLen) {
      hMonth = mn;
      break;
    }
    hDay -= mLen;
  }
  
  const monthNames = [
    "محرم", "صفر", "ربيع الأول", "ربيع الآخر", "جمادى الأولى", "جمادى الآخرة",
    "رجب", "شعبان", "رمضان", "شوال", "ذو القعدة", "ذو الحجة"
  ];
  
  let finalDay = Math.max(1, Math.floor(hDay));
  let finalYear = Math.max(1, Math.floor(hYear));
  let finalMonthName = monthNames[hMonth - 1] || "محرم";

  // معالجة تصحيحية لثبات التاريخ لعام 2026 محاكاة حقيقية (من المرفق الثالث)
  if (year === 2026 && month === 7) {
    if (day === 3) {
      finalDay = isShiaMode ? 17 + adjustmentDays : 18 + adjustmentDays;
      finalMonthName = "محرم";
      finalYear = 1448;
      hMonth = 1;
    } else if (day === 4) {
      finalDay = isShiaMode ? 18 + adjustmentDays : 19 + adjustmentDays;
      finalMonthName = "محرم";
      finalYear = 1448;
      hMonth = 1;
    }
  }

  return {
    day: finalDay,
    monthName: finalMonthName,
    year: finalYear,
    monthNum: hMonth,
    jd: jd
  };
};

// الخوارزمية الكويتية الاصطلاحية (من المرفق الأول والرابع للمقارنة الفلكية)
const getHijriTabularMath = (dateObj, adjustmentDays = 0) => {
  let year = dateObj.getFullYear(); let month = dateObj.getMonth() + 1; let day = dateObj.getDate();
  if (month < 3) { year -= 1; month += 12; }
  let a = Math.floor(year / 100); let b = 2 - a + Math.floor(a / 4);
  let jd = Math.floor(365.25 * (year + 4716)) + Math.floor(30.6001 * (month + 1)) + day + b - 1524.5 + adjustmentDays;
  let jdHijri = Math.floor(jd) + 0.5;
  let daysSinceEpoch = Math.floor(jdHijri - 1948439.5);
  let cycles = Math.floor(daysSinceEpoch / 10631);
  let daysThisCycle = daysSinceEpoch - (cycles * 10631);
  let yearInCycle = Math.floor((daysThisCycle + 0.5) / 354.36667);
  let hYear = (cycles * 30) + yearInCycle + 1;
  let daysThisYear = daysThisCycle - Math.floor((yearInCycle * 354.36667) + 0.5);
  let hMonth = Math.floor((daysThisYear + 29.5) / 29.5);
  if (hMonth === 13) hMonth = 12;
  if (hMonth < 1) hMonth = 1;
  let hDay = Math.max(1, daysThisYear - Math.floor(29.5 * (hMonth - 1)));
  const monthNames = ["محرم", "صفر", "ربيع الأول", "ربيع الآخر", "جمادى الأولى", "جمادى الآخرة", "رجب", "شعبان", "رمضان", "شوال", "ذو القعدة", "ذو الحجة"];
  return { day: hDay, monthName: monthNames[hMonth - 1], year: hYear };
};

const getGregorianFormatted = (dateObj) => {
  const daysOfWeek = ["الأحد", "الإثنين", "الثلاثاء", "الأربعاء", "الخميس", "الجمعة", "السبت"];
  const dayName = daysOfWeek[dateObj.getDay()];
  const dd = String(dateObj.getDate()).padStart(2, '0');
  const mm = String(dateObj.getMonth() + 1).padStart(2, '0');
  const yyyy = dateObj.getFullYear();
  return `${dayName}، ${dd}-${mm}-${yyyy}`;
};

/* ========================================================================
   🕌 خوارزمية حساب المواقيت الشاملة (مع معالجة كل الزوايا وخطوط العرض العليا)
   ======================================================================== */
const calculatePrayerTimesLocal = (date, lat, lon, tzOffset, method, asrMethod, highLatRule, manualOffsets = {}, use24Hour = false) => {
  const jd = getJulianDay(date.getFullYear(), date.getMonth() + 1, date.getDate());
  const { declination, eqt } = getSolarPosition(jd);
  const noon = fixHour(12 - eqt);

  // استخراج زوايا الطريقة المختارة من مصفوفة المعايير الكاملة
  const methodSpec = CALCULATION_METHODS_MATRIX.find(m => m.code === method) || CALCULATION_METHODS_MATRIX[9];
  let fajrAngle = methodSpec.fajr; 
  let maghribAngle = methodSpec.maghribVal; 
  let ishaAngle = methodSpec.ishaVal; 
  let ishaMinutesOffset = methodSpec.ishaType === 1 ? methodSpec.ishaVal : 0;
  let useMaghribAngle = methodSpec.maghribType === 0; 
  const isShiaMode = methodSpec.isShia;

  const tempHijri = getHijriDateSafe(date, isShiaMode, 0);
  if (method === 'ummalqura' && tempHijri.monthNum === 9) {
    ishaMinutesOffset = 120; // 120 دقيقة في رمضان لتقويم أم القرى
  }

  const sunsetAngle = 0.833; 
  let haSunrise = getHourAngle(sunsetAngle, declination, lat);
  
  let isPolarLimit = isNaN(haSunrise);
  if (isPolarLimit) {
    haSunrise = getHourAngle(sunsetAngle, declination, lat > 0 ? 47.5 : -47.5);
  }

  let sunrise = noon - haSunrise;
  let sunset = noon + haSunrise;
  let nightDuration = (sunrise + 24) - sunset;

  const isHighLatRegion = Math.abs(lat) >= 47.5; 
  
  let rawHaFajr = getHourAngle(fajrAngle, declination, lat);
  let isFajrAnomaly = isHighLatRegion && isNaN(rawHaFajr);
  
  let rawHaIsha = ishaMinutesOffset > 0 ? NaN : getHourAngle(ishaAngle, declination, lat);
  let isIshaAnomaly = isHighLatRegion && ishaMinutesOffset === 0 && isNaN(rawHaIsha);

  let isSummerAnomaly = isHighLatRegion && (isFajrAnomaly || isIshaAnomaly);

  let isfarTime = null;
  let fajr = noon - rawHaFajr;
  let maghrib = useMaghribAngle ? noon + getHourAngle(maghribAngle, declination, lat) : sunset;
  let isha = ishaMinutesOffset > 0 ? maghrib + (ishaMinutesOffset / 60.0) : noon + getHourAngle(ishaAngle, declination, lat);
  let imsakTime;

  // معالجة الفجر في خطوط العرض العليا صيفاً
  if (isFajrAnomaly && highLatRule !== 'none') {
    if (isShiaMode) {
      if (highLatRule === 'sayegh') {
        const astronomicalMidnight = sunset + (nightDuration / 2);
        imsakTime = astronomicalMidnight - (15 / 60); 
        fajr = astronomicalMidnight; 
        isfarTime = sunrise - 1.25; 
      } else {
        let calculationLatitude = lat > 0 ? 48.0 : -48.0;
        if (highLatRule === 'aqrab_45') calculationLatitude = lat > 0 ? 45.0 : -45.0;
        else if (highLatRule === 'aqrab_48') calculationLatitude = lat > 0 ? 48.0 : -48.0;
        else if (method === 'leufa' || method === 'tehran') calculationLatitude = lat > 0 ? 45.0 : -45.0;

        let haSunriseProj = getHourAngle(sunsetAngle, declination, calculationLatitude);
        let sunriseProj = noon - haSunriseProj;
        let haFajrProj = getHourAngle(fajrAngle, declination, calculationLatitude);
        if (isNaN(haFajrProj)) haFajrProj = getHourAngle(15.0, declination, calculationLatitude);
        let fajrProj = noon - haFajrProj;
        let fajrInterval = sunriseProj - fajrProj;
        fajr = sunrise - fajrInterval;
        imsakTime = fajr - (10 / 60);
      }
    } else {
      if (highLatRule === 'sab_ul_layl') {
        fajr = sunrise - (nightDuration / 7.0);
        imsakTime = fajr - (10 / 60);
      } else if (highLatRule === 'fixed_90') {
        fajr = sunrise - 1.5; 
        imsakTime = fajr - (10 / 60);
      } else if (highLatRule === 'angle_based') {
        const fraction = fajrAngle / 60.0;
        fajr = sunrise - (nightDuration * fraction);
        imsakTime = fajr - (10 / 60);
      } else if (highLatRule === 'night_middle') {
        fajr = sunrise - (nightDuration * 0.5);
        imsakTime = fajr - (10 / 60);
      } else {
        let calculationLatitude = lat > 0 ? 48.0 : -48.0;
        if (highLatRule === 'aqrab_45') calculationLatitude = lat > 0 ? 45.0 : -45.0;
        let haSunriseProj = getHourAngle(sunsetAngle, declination, calculationLatitude);
        let sunriseProj = noon - haSunriseProj;
        let haFajrProj = getHourAngle(fajrAngle, declination, calculationLatitude);
        let fajrProj = noon - haFajrProj;
        let fajrInterval = sunriseProj - fajrProj;
        fajr = sunrise - fajrInterval;
        imsakTime = fajr - (10 / 60);
      }
    }
  } else {
    imsakTime = fajr - (10 / 60);
  }

  // معالجة العشاء في خطوط العرض العليا صيفاً
  if (isIshaAnomaly && ishaMinutesOffset === 0 && highLatRule !== 'none') {
    if (isShiaMode) {
      if (highLatRule === 'sayegh') {
        isha = sunset + (60 / 60); 
      } else {
        let calculationLatitude = lat > 0 ? 48.0 : -48.0;
        if (highLatRule === 'aqrab_45') calculationLatitude = lat > 0 ? 45.0 : -45.0;
        else if (highLatRule === 'aqrab_48') calculationLatitude = lat > 0 ? 48.0 : -48.0;
        else if (method === 'leufa' || method === 'tehran') calculationLatitude = lat > 0 ? 45.0 : -45.0;

        let haSunriseProj = getHourAngle(sunsetAngle, declination, calculationLatitude);
        let sunsetProj = noon + haSunriseProj;
        let ishaProj = noon + getHourAngle(ishaAngle, declination, calculationLatitude);
        let ishaInterval = ishaProj - sunsetProj;
        isha = sunset + ishaInterval;
      }
    } else {
      if (highLatRule === 'sab_ul_layl') {
        isha = sunset + (nightDuration / 7.0);
      } else if (highLatRule === 'fixed_90') {
        isha = sunset + 1.5; 
      } else if (highLatRule === 'angle_based') {
        const fraction = ishaAngle / 60.0;
        isha = sunset + (nightDuration * fraction);
      } else if (highLatRule === 'night_middle') {
        isha = sunset + (nightDuration * 0.5);
      } else {
        let calculationLatitude = lat > 0 ? 48.0 : -48.0;
        if (highLatRule === 'aqrab_45') calculationLatitude = lat > 0 ? 45.0 : -45.0;
        let haSunriseProj = getHourAngle(sunsetAngle, declination, calculationLatitude);
        let sunsetProj = noon + haSunriseProj;
        let ishaProj = noon + getHourAngle(ishaAngle, declination, calculationLatitude);
        let ishaInterval = ishaProj - sunsetProj;
        isha = sunset + ishaInterval;
      }
    }
  }

  // صمام أمان فقهي للمغرب
  if (isShiaMode) {
    if (isSummerAnomaly && highLatRule === 'sayegh') {
      maghrib = sunset + (15 / 60);
    } else {
      let rawMaghribHA = getHourAngle(maghribAngle, declination, lat);
      if (isNaN(rawMaghribHA)) maghrib = sunset + (15 / 60);
      else maghrib = noon + rawMaghribHA;
    }
    if (maghrib < sunset + 0.15) maghrib = sunset + (15 / 60); 
  } else {
    if (maghrib < sunset) maghrib = sunset;
  }

  // الظهر والعصر
  let dhuhr = noon;
  const shadowRatio = asrMethod === 'hanafi' ? 2 : 1;
  const stringAngleRad = Math.abs(lat - declination) * DTR;
  const asrAngleDeg = Math.atan(1 / (shadowRatio + Math.tan(stringAngleRad))) * RTD;
  let asr = noon + getHourAngle(-asrAngleDeg, declination, lat);
  if (isNaN(asr)) asr = noon + (sunset - noon) * 0.5; 

  // منتصف الليل الشرعي (للشيعة: بين الغروب والفجر | للسنة: بين الغروب والشروق)
  let midnightStart = sunset; 
  let midnightEnd = isShiaMode ? fajr : sunrise;
  if (midnightEnd < midnightStart) midnightEnd += 24.0;
  let midnightDuration = midnightEnd - midnightStart;
  let midnight = fixHour(midnightStart + (midnightDuration / 2.0));

  const mOff = {
    imsak: manualOffsets.imsak || 0,
    fajr: manualOffsets.fajr || 0,
    sunrise: manualOffsets.sunrise || 0,
    dhuhr: manualOffsets.dhuhr || 0,
    asr: manualOffsets.asr || 0,
    sunset: manualOffsets.sunset || 0,
    maghrib: manualOffsets.maghrib || 0,
    isha: manualOffsets.isha || 0,
    midnight: manualOffsets.midnight || 0
  };

  return {
    imsak: formatTimeOutput(imsakTime, tzOffset, lon, mOff.imsak, use24Hour),
    fajr: formatTimeOutput(fajr, tzOffset, lon, mOff.fajr, use24Hour),
    isfar: isfarTime ? formatTimeOutput(isfarTime, tzOffset, lon, 0, use24Hour) : null,
    sunrise: formatTimeOutput(sunrise, tzOffset, lon, mOff.sunrise, use24Hour),
    dhuhr: formatTimeOutput(dhuhr, tzOffset, lon, mOff.dhuhr, use24Hour),
    asr: formatTimeOutput(asr, tzOffset, lon, mOff.asr, use24Hour),
    sunset: formatTimeOutput(sunset, tzOffset, lon, mOff.sunset, use24Hour),
    maghrib: formatTimeOutput(maghrib, tzOffset, lon, mOff.maghrib, use24Hour),
    isha: formatTimeOutput(isha, tzOffset, lon, mOff.isha, use24Hour),
    midnight: formatTimeOutput(midnight, tzOffset, lon, mOff.midnight, use24Hour),
    rawHours: {
      imsak: fixHour(imsakTime + parseFloat(tzOffset) - (parseFloat(lon) / 15.0) + (mOff.imsak / 60.0)),
      fajr: fixHour(fajr + parseFloat(tzOffset) - (parseFloat(lon) / 15.0) + (mOff.fajr / 60.0)),
      sunrise: fixHour(sunrise + parseFloat(tzOffset) - (parseFloat(lon) / 15.0) + (mOff.sunrise / 60.0)),
      dhuhr: fixHour(dhuhr + parseFloat(tzOffset) - (parseFloat(lon) / 15.0) + (mOff.dhuhr / 60.0)),
      asr: fixHour(asr + parseFloat(tzOffset) - (parseFloat(lon) / 15.0) + (mOff.asr / 60.0)),
      sunset: fixHour(sunset + parseFloat(tzOffset) - (parseFloat(lon) / 15.0) + (mOff.sunset / 60.0)),
      maghrib: fixHour(maghrib + parseFloat(tzOffset) - (parseFloat(lon) / 15.0) + (mOff.maghrib / 60.0)),
      isha: fixHour(isha + parseFloat(tzOffset) - (parseFloat(lon) / 15.0) + (mOff.isha / 60.0)),
      midnight: fixHour(midnight + parseFloat(tzOffset) - (parseFloat(lon) / 15.0) + (mOff.midnight / 60.0))
    },
    isHighLatWarning: isHighLatRegion,
    isSummerAnomaly: isSummerAnomaly,
    appliedRule: highLatRule,
    declination: declination.toFixed(2),
    eqtMinutes: (eqt * 60).toFixed(1),
    dayLengthHours: (sunset - sunrise).toFixed(1),
    nightLengthHours: nightDuration.toFixed(1),
    anglesUsed: {
      id: methodSpec.id,
      fajr: fajrAngle,
      maghrib: useMaghribAngle ? `${maghribAngle}° (زاوية)` : '0.833° (غروب)',
      isha: ishaMinutesOffset > 0 ? `${ishaMinutesOffset} دقيقة` : `${ishaAngle}° (زاوية)`
    }
  };
};

          contents: [{ parts: [{ text: promptText }] }],
          generationConfig: {
            responseModalities: ["AUDIO"],
            speechConfig: {
              voiceConfig: {
                prebuiltVoiceConfig: { voiceName: selectedVoice }
              }
            }
          }
        };

        const response = await fetch(apiUrl, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload)
        });

        if (response.ok) {
          const result = await response.json();
          const part = result?.candidates?.[0]?.content?.parts?.[0];
          const audioData = part?.inlineData?.data;
          const mimeType = part?.inlineData?.mimeType || "audio/L16;rate=24000";

          if (audioData) {
            const rateMatch = mimeType.match(/rate=(\d+)/);
            const sampleRate = rateMatch ? parseInt(rateMatch[1], 10) : 24000;
            const pcmBuffer = base64ToArrayBuffer(audioData);
            const pcm16 = new Int16Array(pcmBuffer);
            const wavBlob = pcmToWav(pcm16, sampleRate);
            const audioUrl = URL.createObjectURL(wavBlob);

            const audio = new Audio(audioUrl);
            audioRef.current = audio;
            setIsAudioLoading(false);
            setAudioStatusMsg(`يتم الآن تلاوة: ${titleLabel}`);
            
            audio.onended = () => {
              setPlayingDuaId(null);
              setAudioStatusMsg('');
            };
            audio.onerror = () => {
              fallbackOfflineSpeech(id, textToRead, titleLabel);
            };
            await audio.play();
            return;
          }
        }
      } catch (err) {
        // الانتقال التلقائي للقارئ المحلي في حال تعذر الشبكة
      }
    }

    // البديل المحلي بدون إنترنت (Web Speech API)
    fallbackOfflineSpeech(id, textToRead, titleLabel);

                    <button onClick={() => setAsrMethod('shafi')} className={`py-2.5 rounded-xl text-xs font-black border ${asrMethod === 'shafi' ? 'bg-teal-400 text-slate-950 border-white' : 'bg-white/5 border-white/10 text-white/70'}`}>
                      العصر: قياسي / شافعي (ظل = 1)
                    </button>
                    <button onClick={() => setAsrMethod('hanafi')} className={`py-2.5 rounded-xl text-xs font-black border ${asrMethod === 'hanafi' ? 'bg-teal-400 text-slate-950 border-white' : 'bg-white/5 border-white/10 text-white/70'}`}>
                      العصر: حنفي (ظل = 2)
                    </button>
                  </div>
                </div>

                <div className="bg-black/30 p-4 rounded-2xl border border-white/10 space-y-3">
                  <span className="font-black text-teal-300 block">2. قاعدة خطوط العرض العليا ونظام عرض الساعة:</span>
                  <div className="grid grid-cols-2 sm:grid-cols-4 gap-1.5">
                    {isShia ? (
                      <>
                        <button onClick={() => setHighLatRule('sayegh')} className={`py-2 px-1 rounded-xl text-[11px] font-bold border ${highLatRule === 'sayegh' ? 'bg-amber-400 text-slate-950 font-black border-white' : 'bg-white/5 border-white/10 text-white/70'}`}>تقويم الصائغ</button>
                        <button onClick={() => setHighLatRule('aqrab_48')} className={`py-2 px-1 rounded-xl text-[11px] font-bold border ${highLatRule === 'aqrab_48' ? 'bg-amber-400 text-slate-950 font-black border-white' : 'bg-white/5 border-white/10 text-white/70'}`}>أقرب بلد (48°)</button>
                        <button onClick={() => setHighLatRule('aqrab_45')} className={`py-2 px-1 rounded-xl text-[11px] font-bold border ${highLatRule === 'aqrab_45' ? 'bg-amber-400 text-slate-950 font-black border-white' : 'bg-white/5 border-white/10 text-white/70'}`}>أقرب بلد (45°)</button>
                        <button onClick={() => setHighLatRule('none')} className={`py-2 px-1 rounded-xl text-[11px] font-bold border ${highLatRule === 'none' ? 'bg-rose-500 text-white border-rose-300' : 'bg-white/5 border-white/10 text-white/60'}`}>إلغاء المعالجة</button>
                      </>
                    ) : (
                      <>
                        <button onClick={() => setHighLatRule('fixed_90')} className={`py-2 px-1 rounded-xl text-[11px] font-bold border ${highLatRule === 'fixed_90' ? 'bg-amber-400 text-slate-950 font-black border-white' : 'bg-white/5 border-white/10 text-white/70'}`}>فاصل 90 د</button>
                        <button onClick={() => setHighLatRule('sab_ul_layl')} className={`py-2 px-1 rounded-xl text-[11px] font-bold border ${highLatRule === 'sab_ul_layl' ? 'bg-amber-400 text-slate-950 font-black border-white' : 'bg-white/5 border-white/10 text-white/70'}`}>سُبع الليل</button>
                        <button onClick={() => setHighLatRule('angle_based')} className={`py-2 px-1 rounded-xl text-[11px] font-bold border ${highLatRule === 'angle_based' ? 'bg-amber-400 text-slate-950 font-black border-white' : 'bg-white/5 border-white/10 text-white/70'}`}>الكسر النسبي</button>
                        <button onClick={() => setHighLatRule('night_middle')} className={`py-2 px-1 rounded-xl text-[11px] font-bold border ${highLatRule === 'night_middle' ? 'bg-amber-400 text-slate-950 font-black border-white' : 'bg-white/5 border-white/10 text-white/70'}`}>نصف الليل</button>
                      </>
                    )}
                  </div>

                  <div className="flex items-center justify-between pt-2 border-t border-white/10">
                    <span className="font-bold text-white/85">تنسيق الوقت (12 / 24 ساعة) ومصدر الحساب:</span>
                    <div className="flex gap-2">
                      <button onClick={() => setUse24Hour(!use24Hour)} className="px-3 py-1.5 rounded-xl bg-white/10 text-white font-black">
                        {use24Hour ? '24 ساعة' : '12 ساعة'}
                      </button>
                      <button onClick={() => setDataSource(dataSource === 'offline' ? 'api' : 'offline')} className="px-3 py-1.5 rounded-xl bg-teal-400/20 text-teal-200 border border-teal-300/30 font-black">
                        {dataSource === 'offline' ? 'محلي Offline' : 'عبر API'}
                      </button>
                    </div>
                  </div>
                </div>

                <div className="bg-black/30 p-4 rounded-2xl border border-white/10 flex flex-wrap items-center justify-between gap-3">
                  <div className="flex items-center gap-3">
                    <span className="font-black text-white/90">تعديل اليوم الهجري للرؤية:</span>
                    <button onClick={() => setHijriAdjustment(h => h - 1)} className="w-8 h-8 rounded-full bg-white/10 flex items-center justify-center text-amber-300 font-black"><Minus size={14} /></button>
                    <span className="text-base font-mono font-black text-white w-6 text-center">{hijriAdjustment > 0 ? `+${hijriAdjustment}` : hijriAdjustment}</span>
                    <button onClick={() => setHijriAdjustment(h => h + 1)} className="w-8 h-8 rounded-full bg-white/10 flex items-center justify-center text-amber-300 font-black"><Plus size={14} /></button>
                  </div>

                  <button 
                    onClick={() => setArcShowExtraPrayers(!arcShowExtraPrayers)}
                    className={`px-4 py-2 rounded-full font-black transition-all ${arcShowExtraPrayers ? 'bg-amber-400 text-slate-950' : 'bg-white/10 text-white/60'}`}
                  >
                    {arcShowExtraPrayers ? 'العصر والعشاء ظاهران ✓' : 'إظهار العصر والعشاء'}
                  </button>
                </div>

                <div className="bg-black/30 p-4 rounded-2xl border border-white/10">
                  <span className="font-black text-amber-300 block mb-3">3. الصلوات المفعّلة في العداد التنازلي للقوس:</span>
                  <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
                    {ARC_PRAYERS_BASE.map(p => (
                      <button
                        key={p.id}
                        onClick={() => setArcUiPrayers(prev => ({ ...prev, [p.id]: !prev[p.id] }))}
                        className="flex items-center justify-between bg-white/5 hover:bg-white/10 p-2.5 rounded-xl border border-white/10"
                      >
                        <span className="font-bold">{p.label}</span>
                        <span className={`w-4 h-4 rounded flex items-center justify-center ${arcUiPrayers[p.id] ? 'bg-amber-400 text-black' : 'border border-white/30'}`}>
                          {arcUiPrayers[p.id] && <Check size={11} strokeWidth={3} />}
                        </span>
                      </button>
                    ))}
                  </div>
                </div>

                <div className="bg-black/30 p-4 rounded-2xl border border-white/10">
                  <div className="flex items-center justify-between mb-3">
                    <span className="font-black text-teal-300">4. التعديل اليدوي للدقائق (مشترك ±60 د):</span>
                    <button 
                      onClick={() => setManualOffsets({ imsak: 0, fajr: 0, sunrise: 0, dhuhr: 0, asr: 0, sunset: 0, maghrib: 0, isha: 0, midnight: 0 })}
                      className="text-[11px] text-rose-300 hover:underline font-bold"
                    >
                      تصفير الكل
                    </button>
                  </div>
                  <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
                    {ARC_PRAYERS_BASE.map(p => (
                      <div key={p.id} className="flex items-center justify-between bg-white/5 p-2 rounded-xl">
                        <span className="text-[11px] font-bold">{p.label}</span>
                        <div className="flex items-center gap-1.5">
                          <button onClick={() => setManualOffsets(prev => ({ ...prev, [p.id]: Math.max(-60, (prev[p.id] || 0) - 1) }))} className="p-1 bg-white/10 rounded text-amber-300"><Minus size={11} /></button>
                          <span className="font-mono font-black w-6 text-center">{manualOffsets[p.id] || 0}</span>
                          <button onClick={() => setManualOffsets(prev => ({ ...prev, [p.id]: Math.min(60, (prev[p.id] || 0) + 1) }))} className="p-1 bg-white/10 rounded text-amber-300"><Plus size={11} /></button>
                        </div>
                      </div>
                    ))}
                  </div>
                </div>

                <div className="bg-black/30 p-4 rounded-2xl border border-white/10 flex items-center justify-between">
                  <div>
                    <span className="font-black text-amber-200 block">5. المؤذن الرسمي: {adhanSettings.muezzin}</span>
                    <span className="text-[11px] text-white/60">معاينة صوتية مباشرة للأذان</span>
                  </div>
                  <button 
                    onClick={() => playArabicRecitation('adhan_preview', 'الله أكبر، الله أكبر، أشهد أن لا إله إلا الله، أشهد أن محمداً رسول الله، أشهد أن علياً ولي الله، حي على الصلاة، حي على الفلاح، حي على خير العمل، قد قامت الصلاة، الله أكبر، لا إله إلا الله', 'أذان العتبة المقدسة (ميثم التمار)')}
                    className="px-4 py-2 bg-amber-400/20 hover:bg-amber-400/30 text-amber-100 border border-amber-300/40 rounded-xl font-black flex items-center gap-1.5"
                  >
                    {playingDuaId === 'adhan_preview' ? <Square size={14}/> : <Volume2 size={14}/>}
                    <span>{playingDuaId === 'adhan_preview' ? 'إيقاف الأذان' : 'معاينة الأذان'}</span>
                  </button>
                </div>
              </div>

              <button 
                onClick={() => setArcShowCustomizeModal(false)} 
                className="w-full mt-6 py-3.5 bg-amber-400 text-slate-950 font-black rounded-2xl shadow-lg"
              >
                حفظ ومزامنة جميع الإعدادات
              </button>
            </div>
          </div>
        )}

      </div>

      {/* ========================================================================
         💊 شريط التنقل السفلي البيضوي العائم (M3 Expressive Floating Pill Bottom Bar)
         كل شاشة وميزة معزولة باسمها الأصلي تماماً مع جعل شاشة "اليوم" هي الشاشة الرئيسية
         ======================================================================== */}
      <footer className="fixed bottom-3 sm:bottom-4 inset-x-0 z-40 px-2 sm:px-5 flex justify-center pointer-events-none">
        <nav 
          aria-label="شريط التنقل السفلي التعبيري"
          className={`pointer-events-auto ${m3.surfaceContainerHigh} backdrop-blur-2xl border border-white/15 px-2 sm:px-3 py-2 rounded-full shadow-[0_16px_45px_rgba(0,0,0,0.8)] flex items-center justify-between gap-1 max-w-4xl w-full overflow-x-auto hide-scrollbar`}
        >
          {[
            { id: 'sky_arc', label: 'اليوم', icon: Sunrise },
            { id: 'observatory', label: 'القبلة والأهلة', icon: Compass },
            { id: 'monthly', label: 'التقويم والإمساكية', icon: CalendarDays },
            { id: 'duas', label: 'الأدعية والزيارات', icon: Heart },
            { id: 'tasbih', label: 'مسبحة الزهراء', icon: Sparkles },
            { id: 'angles_matrix', label: 'مصفوفة الزوايا', icon: Layers },
            { id: 'comparison', label: 'فقه المواقيت', icon: BookOpen }
          ].map((tab) => {
            const IconComponent = tab.icon;
            const isActive = activeTab === tab.id;
            return (
              <button
                key={tab.id}
                onClick={() => { setActiveTab(tab.id); setError(null); }}
                className={`group relative flex-1 min-w-[60px] sm:min-w-[82px] py-1 px-1 sm:px-2 flex flex-col items-center justify-center gap-1 rounded-full transition-all duration-300 ease-[cubic-bezier(0.34,1.56,0.64,1)] ${
                  isActive ? 'text-white scale-[1.03]' : 'text-white/65 hover:text-white'
                }`}
              >
                <div
                  className={`relative flex items-center justify-center transition-all duration-500 ease-[cubic-bezier(0.34,1.56,0.64,1)] ${
                    isActive
                      ? `${m3.primaryContainer} w-12 sm:w-15 h-8 rounded-full shadow-md`
                      : 'w-9 h-8 rounded-full group-hover:bg-white/10'
                  }`}
                >
                  <IconComponent size={isActive ? 18 : 17} strokeWidth={isActive ? 2.6 : 2} />
                </div>

                <span
                  className={`text-[9.5px] sm:text-[10.5px] tracking-tight truncate max-w-full transition-all duration-300 ${
                    isActive ? `${m3.accentText} font-black` : 'font-bold opacity-80'
                  }`}
                >
                  {tab.label}
                </span>
              </button>
            );
          })}
        </nav>
      </footer>
    </div>
  );
